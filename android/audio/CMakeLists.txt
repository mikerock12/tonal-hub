cmake_minimum_required(VERSION 3.28)
project(tonal_audio LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(EIGEN_BUILD_DOC OFF CACHE BOOL "" FORCE)
set(EIGEN_BUILD_PKGCONFIG OFF CACHE BOOL "" FORCE)
get_filename_component(ROOT "../.." ABSOLUTE)
set(DEPS "${ROOT}/out/audio-deps")
set(MOBILE "${ROOT}/out/android/audio-deps")
set(CODECS "${DEPS}/iamf-build/dep_codecs")
foreach(dep absl eigen googletest pffft)
  string(TOUPPER "${dep}" upper)
  set(FETCHCONTENT_SOURCE_DIR_${upper} "${DEPS}/iamf-build/_deps/${dep}-src" CACHE PATH "" FORCE)
endforeach()
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_CXXLIBS OFF CACHE BOOL "" FORCE)
set(WITH_OGG OFF CACHE BOOL "" FORCE)
set(INSTALL_MANPAGES OFF CACHE BOOL "" FORCE)
add_subdirectory("${CODECS}/dep_opus-prefix/src/dep_opus" opus EXCLUDE_FROM_ALL)
add_subdirectory("${CODECS}/dep_flac-prefix/src/dep_flac" flac EXCLUDE_FROM_ALL)
# Keep the desktop source/cache untouched; Android gets its own FDK source copy.
add_subdirectory("${MOBILE}/fdk-aac" aac EXCLUDE_FROM_ALL)
set(OAR_BUILD_SHARED_LIBRARY OFF CACHE BOOL "" FORCE)
set(OAR_ENABLE_BINAURALIZER ON CACHE BOOL "" FORCE)
set(OAR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(IAMF "${DEPS}/libiamf-src/code")
add_subdirectory("${IAMF}/dep_external/src/oar" oar EXCLUDE_FROM_ALL)
target_compile_options(oar PRIVATE "-include${CMAKE_CURRENT_SOURCE_DIR}/android_time_compat.h")
file(GLOB_RECURSE IAMF_SOURCES CONFIGURE_DEPENDS "${IAMF}/src/*.c" "${IAMF}/dep_external/src/wav/*.c")
add_library(iamf STATIC ${IAMF_SOURCES})
target_compile_definitions(iamf PRIVATE CONFIG_OPUS_CODEC CONFIG_AAC_CODEC CONFIG_FLAC_CODEC FLAC__NO_DLL)
target_include_directories(iamf PUBLIC "${IAMF}/include" PRIVATE
  "${IAMF}/dep_codecs/include" "${IAMF}/dep_external/include"
  "${IAMF}/dep_external/src/oar/include" "${IAMF}/dep_external/src/oar/src/common" "${IAMF}/dep_external/src/oar/src/limiter"
  "${IAMF}/src/iamf_dec" "${IAMF}/src/iamf_dec/codec" "${IAMF}/src/iamf_dec/codec/opus"
  "${IAMF}/src/iamf_dec/codec/aac" "${IAMF}/src/iamf_dec/codec/flac" "${IAMF}/src/iamf_dec/common" "${IAMF}/src/iamf_dec/obu")
target_link_libraries(iamf PRIVATE opus FLAC fdk-aac oar m)
set(LIBTYPE SHARED CACHE STRING "" FORCE)
foreach(option ALSOFT_UTILS ALSOFT_EXAMPLES ALSOFT_TESTS ALSOFT_INSTALL ALSOFT_BACKEND_OPENSL ALSOFT_BACKEND_OBOE)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(ALSOFT_EMBED_HRTF_DATA ON CACHE BOOL "" FORCE)
set(ALSOFT_ENABLE_MODULES OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/openal-soft" openal EXCLUDE_FROM_ALL)
# Bionic's C11 thread API is newer than minSdk 28; use the POSIX implementation.
target_compile_definitions(OpenAL PRIVATE __STDC_NO_THREADS__=1)
target_compile_definitions(alsoft.common PRIVATE __STDC_NO_THREADS__=1)
# NDK 27 libc++ cannot instantiate views::drop on this FlexArray type.
# Compile a generated one-line compatibility patch; preserve the desktop source.
file(READ "${DEPS}/openal-soft/al/auxeffectslot.cpp" OPENAL_AUX)
string(REPLACE "std::ranges::fill(*newarray | std::views::drop(newcount), nullptr);"
  "std::fill(newarray->begin() + newcount, newarray->end(), nullptr);" OPENAL_AUX "${OPENAL_AUX}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/openal_auxeffectslot.cpp" "${OPENAL_AUX}")
set_source_files_properties("${DEPS}/openal-soft/al/auxeffectslot.cpp" DIRECTORY "${DEPS}/openal-soft" PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(OpenAL PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/openal_auxeffectslot.cpp")
target_include_directories(OpenAL PRIVATE "${DEPS}/openal-soft/al")
add_library(ebur128 STATIC "${MOBILE}/libebur128-1.2.6/ebur128/ebur128.c")
target_include_directories(ebur128 PUBLIC "${MOBILE}/libebur128-1.2.6/ebur128")
add_library(tonal_audio SHARED audio_plugin.cpp)
target_include_directories(tonal_audio PRIVATE "${MOBILE}/${ANDROID_ABI}/ffmpeg/include" "${DEPS}/openal-soft/include")
foreach(lib avfilter avformat avcodec swresample avutil)
  add_library(${lib} SHARED IMPORTED)
  set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${MOBILE}/${ANDROID_ABI}/ffmpeg/lib/lib${lib}.so")
endforeach()
target_link_libraries(tonal_audio PRIVATE avfilter avformat avcodec swresample avutil OpenAL iamf ebur128 log android m)
target_link_options(tonal_audio PRIVATE "-Wl,-z,max-page-size=16384")
