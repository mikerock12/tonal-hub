cmake_minimum_required(VERSION 3.21)
project(TonalStories LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(TONAL_LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../out/story-deps/llama" CACHE PATH "Verified llama.cpp sources")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(flag LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_MTMD LLAMA_OPENSSL GGML_NATIVE GGML_OPENMP GGML_LLAMAFILE GGML_CUDA GGML_VULKAN GGML_METAL GGML_BACKEND_DL)
 set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
option(TONAL_STORY_ARM_FAST "Optional ARM64 reader, loaded only after runtime HWCAP checks" OFF)
if(TONAL_STORY_ARM_FAST)
 if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Accelerated reader requires Android ARM64")
 endif()
 set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+i8mm+fp16" CACHE STRING "" FORCE)
endif()
add_subdirectory("${TONAL_LLAMA_SOURCE}" llama EXCLUDE_FROM_ALL)
add_library(tonal_story_core STATIC LocalStoryModel.cpp)
target_compile_features(tonal_story_core PRIVATE cxx_std_20)
target_include_directories(tonal_story_core PUBLIC .)
target_link_libraries(tonal_story_core PRIVATE llama)
if(TONAL_STORY_ARM_FAST)
 target_compile_definitions(tonal_story_core PRIVATE TONAL_STORY_ARM_FAST=1)
endif()
if(MSVC)
 target_compile_options(tonal_story_core PRIVATE /utf-8)
endif()
option(TONAL_STORY_CORE_ONLY "Build portable inference guards without the UI helper" OFF)
if(TONAL_STORY_CORE_ONLY OR NOT ANDROID)
 enable_testing()
 add_executable(TonalStoryGuards ../../tests/stories/ModelGuards.cpp)
 target_link_libraries(TonalStoryGuards PRIVATE tonal_story_core)
 if(MSVC)
  target_compile_options(TonalStoryGuards PRIVATE /utf-8)
 endif()
 add_test(NAME local_model_guards COMMAND TonalStoryGuards)
endif()
if(TONAL_STORY_CORE_ONLY)
 # The portable CI builds guard checks without Qt or model weights.
elseif(ANDROID)
 add_library(tonal_story SHARED StoryJni.cpp)
 target_link_libraries(tonal_story PRIVATE tonal_story_core llama)
 set_target_properties(tonal_story PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
 if(TONAL_STORY_ARM_FAST)
  target_compile_definitions(tonal_story PRIVATE TONAL_STORY_ARM_FAST=1)
  target_link_options(tonal_story PRIVATE -Wl,-Bsymbolic)
  set_target_properties(tonal_story PROPERTIES OUTPUT_NAME tonal_story_fast)
 endif()
 target_link_options(tonal_story PRIVATE -Wl,-z,max-page-size=16384)
else()
 find_package(Qt6 6.8 REQUIRED COMPONENTS Core)
 add_executable(TonalStories StoryMain.cpp)
 target_link_libraries(TonalStories PRIVATE tonal_story_core llama Qt6::Core)
 if(MSVC)
  target_compile_options(TonalStories PRIVATE /utf-8)
 endif()
 set_target_properties(TonalStories PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endif()
